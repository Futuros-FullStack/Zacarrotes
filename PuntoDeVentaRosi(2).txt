TABLAS EN BRUTO

CREATE TABLE proveedor (idproveedor INT GENERATED ALWAYS AS IDENTITY primary key, nombreproveedor text UNIQUE NOT NULL, direccion text UNIQUE NOT NULL, telefono VARCHAR(10),
CONSTRAINT check_telefono CHECK (telefono ~'^[0-9]{10}$')
);


CREATE TABLE producto (idproducto INT GENERATED ALWAYS AS IDENTITY primary key, nombreproducto text NOT NULL, marca text NOT NULL, precio DECIMAL(10,2), caducidad DATE NOT NULL, cantidad int NOT NULL, imagen_url text, idproveedor int NOT NULL,
CONSTRAINT fk_proveedor_producto FOREIGN KEY (idproveedor) REFERENCES proveedor(idproveedor) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT check_fecha CHECK (caducidad >= CURRENT_DATE),
CONSTRAINT check_cantidad CHECK (cantidad >= 0)
);


CREATE TABLE cliente (idcliente INT GENERATED ALWAYS AS IDENTITY primary key, nombre text NOT NULL DEFAULT 'Desconocido'
);


CREATE TABLE venta (idventa INT GENERATED ALWAYS AS IDENTITY primary key, fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP, idcliente int NOT NULL DEFAULT 1, total DECIMAL(10,2),
CONSTRAINT fk_cliente_venta FOREIGN KEY (idcliente) REFERENCES cliente(idcliente) ON DELETE RESTRICT ON UPDATE CASCADE
);


CREATE TABLE detalleventa (idventa int NOT NULL, idproducto int NOT NULL, cantidad int NOT NULL, total DECIMAL(10,2), subtotal DECIMAL(10,2), primary key(idventa, idproducto),
CONSTRAINT fk_venta_detalle FOREIGN KEY (idventa) REFERENCES venta(idventa) ON DELETE RESTRICT ON UPDATE CASCADE,
CONSTRAINT fk_producto_detalle FOREIGN KEY (idproducto) REFERENCES producto(idproducto) ON DELETE RESTRICT ON UPDATE CASCADE
);

VISTAS

(PROVEEDORES Y PRODUCTOS)

CREATE OR REPLACE VIEW v_catalogo_completo AS
SELECT
p.idproducto,
p.nombreproducto AS producto,
p.marca,
p.precio,
p.caducidad,
p.cantidad,
p.imagen_url,
pr.nombreproveedor AS proveedor,
pr.telefono AS tel_proveedor
FROM producto p
JOIN proveedor pr ON pr.idproveedor = p.idproveedor;

(TICKET COMPLETO)

CREATE OR REPLACE VIEW v_ticket AS
SELECT
v.idventa,
v.fecha,
v.total,
c.nombre AS nombre_cliente,
p.nombreproducto AS producto,
p.marca,
dv.cantidad AS cantidad_comprada,
dv.subtotal
FROM venta v
JOIN cliente c ON c.idcliente = v.idcliente
JOIN detalleventa dv ON v.idventa = dv.idventa
JOIN producto p ON p.idproducto = dv.idproducto;


(ALERTA DE INVENTARIO)

CREATE OR REPLACE VIEW v_alerta_inventario AS
SELECT
p.nombreproducto AS producto,
p.marca,
p.cantidad,
p.caducidad
FROM producto p
WHERE cantidad <= 5
OR caducidad <= CURRENT_DATE + INTERVAL '30 days';

PROCEDURES PARA VENTA Y DETALLEVENTA

(ABASTECIMIENTO DE INVENTARIO)

CREATE OR REPLACE PROCEDURE p_abastecer(
p_idproducto INT,
p_cantidad_agregada INT,
p_nueva_caducidad DATE
)
LANGUAGE plpgsql
AS $$
BEGIN
UPDATE producto SET cantidad = p_cantidad_agregada + cantidad, caducidad = p_nueva_caducidad WHERE idproducto = p_idproducto;
COMMIT;
END;
$$;


(REGISTRO DE VENTA)

CREATE OR REPLACE PROCEDURE p_registro_venta(
p_idcliente INT,
p_idproducto INT,
p_cantidadcomprada INT
)
LANGUAGE plpgsql
AS $$
DECLARE
v_precio_unidad DECIMAL(10,2);
v_idventa INT;
v_subtotal DECIMAL(10,2);
BEGIN
SELECT precio INTO v_precio_unidad FROM producto WHERE idproducto = p_idproducto;
v_subtotal := v_precio_unidad * p_cantidadcomprada;
INSERT INTO venta (idcliente, total) VALUES (p_idcliente, v_subtotal)
RETURNING idventa INTO v_idventa;
INSERT INTO detalleventa (idventa, idproducto, cantidad, subtotal, total) 
VALUES (v_idventa, p_idproducto, p_cantidadcomprada, v_subtotal, v_subtotal);
UPDATE producto SET cantidad = cantidad - p_cantidadcomprada WHERE idproducto = p_idproducto;
COMMIT;
END;
$$;


(ELIMINAR VENTA)

CREATE OR REPLACE PROCEDURE p_eliminar_venta(
p_idventa INT
)
LANGUAGE plpgsql
AS $$
BEGIN
UPDATE producto p SET cantidad = p.cantidad + dv.cantidad FROM detalleventa dv
WHERE p.idproducto = dv.idproducto
AND p_idventa = dv.idventa;
DELETE FROM detalleventa WHERE idventa = p_idventa;
DELETE FROM venta WHERE idventa = p_idventa;
COMMIT;
END;
$$;


(EDITAR VENTA)

CREATE OR REPLACE PROCEDURE p_editar_venta(
p_idventa INT,
p_idproducto INT,
p_cantidad_nueva INT
)
LANGUAGE plpgsql
AS $$
DECLARE
v_cantidad_anterior INT;
v_diferencia INT;
v_precio_unidad DECIMAL(10,2);
v_nuevo_subtotal DECIMAL(10,2);
BEGIN
SELECT cantidad INTO v_cantidad_anterior FROM detalleventa WHERE idventa = p_idventa
AND idproducto = p_idproducto;
v_diferencia := p_cantidad_nueva - v_cantidad_anterior;
UPDATE producto SET cantidad = cantidad - v_diferencia WHERE idproducto = p_idproducto;
SELECT precio INTO v_precio_unidad FROM producto WHERE idproducto = p_idproducto;
v_nuevo_subtotal := v_precio_unidad * p_cantidad_nueva;
UPDATE detalleventa SET cantidad = p_cantidad_nueva, subtotal = v_nuevo_subtotal, total = v_nuevo_subtotal
WHERE idventa = p_idventa
AND idproducto = p_idproducto;
UPDATE venta SET total = (SELECT SUM(subtotal) FROM detalleventa WHERE idventa = p_idventa) WHERE idventa = p_idventa;
COMMIT;
END;
$$;

)

END;
$$;


(INSERTAR PROVEEDOR)

CREATE OR REPLACE PROCEDURE p_insertar_proveedor(
p_nombre TEXT,
p_direccion TEXT,
p_telefono VARCHAR
)
LANGUAGE plpgsql
AS $$
BEGIN
INSERT INTO proveedor (nombreproveedor, direccion, telefono) VALUES (p_nombre, p_direccion, p_telefono);
COMMIT;
END;
$$;


(EDITAR PROVEEDOR)

CREATE OR REPLACE PROCEDURE p_editar_proveedor(
p_idproveedor INT,
p_nombre TEXT,
p_direccion TEXT,
p_telefono VARCHAR
)
LANGUAGE plpgsql
AS $$
BEGIN
UPDATE proveedor SET nombreproveedor = p_nombre, 
direccion = p_direccion,
telefono = p_telefono
WHERE idproveedor = p_idproveedor;
COMMIT;
END;
$$;


(INSERTAR PRODUCTO)

CREATE OR REPLACE PROCEDURE p_insertar_producto(
p_nombre TEXT,
p_marca TEXT,
p_precio DECIMAL(10,2),
p_caducidad DATE,
p_cantidad INT,
p_imagen_url TEXT,
p_idproveedor INT
)
LANGUAGE plpgsql
AS $$
BEGIN
INSERT INTO producto (nombreproducto, marca, precio, caducidad, cantidad, imagen_url, idproveedor)
VALUES (p_nombre, p_marca, p_precio, p_caducidad, p_cantidad, p_imagen_url, p_idproveedor);
COMMIT;
END;
$$;



(EDITAR PRODUCTO)

CREATE OR REPLACE PROCEDURE p_editar_producto(
    p_idproducto INT,
    p_nombre TEXT,
    p_marca TEXT,
    p_precio DECIMAL(10,2),
    p_imagen_url TEXT
)
LANGUAGE plpgsql
AS $$
BEGIN
        UPDATE producto 
    SET nombreproducto = p_nombre,
        marca = p_marca,
        precio = p_precio,
        imagen_url = p_imagen_url
    WHERE idproducto = p_idproducto;
    
    COMMIT;
END;
$$;
